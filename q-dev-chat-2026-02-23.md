### Step 4: Service Class

**File: `includes/GoogleCalendarService.php`**
```php
<?php
namespace MusicAll\Services;

use Google_Client;
use Google_Service_Calendar;
use Google_Service_Calendar_Event;

class GoogleCalendarService {
    private $client;
    private $service;
    
    public function __construct($userId) {
        $this->client = new Google_Client();
        $this->client->setAuthConfig(__DIR__ . '/../config/credentials.json');
        $this->client->setRedirectUri(config('google_calendar.redirect_uri'));
        $this->client->setScopes(config('google_calendar.scopes'));
        
        // Carica token salvato per l'utente
        $token = $this->getStoredToken($userId);
        if ($token) {
            $this->client->setAccessToken($token);
            if ($this->client->isAccessTokenExpired()) {
                $this->client->fetchAccessTokenWithRefreshToken();
                $this->saveToken($userId, $this->client->getAccessToken());
            }
        }
        
        $this->service = new Google_Service_Calendar($this->client);
    }
    
    public function syncLezione($lezione) {
        $event = new Google_Service_Calendar_Event([
            'summary' => "{$lezione->materia->nome} - {$lezione->allievo->nome_completo}",
            'location' => $lezione->aula->nome,
            'description' => "Docente: {$lezione->docente->nome_completo}",
            'start' => [
                'dateTime' => $lezione->data_ora_inizio->toRfc3339String(),
                'timeZone' => 'Europe/Rome',
            ],
            'end' => [
                'dateTime' => $lezione->data_ora_fine->toRfc3339String(),
                'timeZone' => 'Europe/Rome',
            ],
            'reminders' => [
                'useDefault' => false,
                'overrides' => [
                    ['method' => 'popup', 'minutes' => 30],
                ],
            ],
        ]);
        
        if ($lezione->google_event_id) {
            // Aggiorna evento esistente
            return $this->service->events->update('primary', $lezione->google_event_id, $event);
        } else {
            // Crea nuovo evento
            $result = $this->service->events->insert('primary', $event);
            $lezione->google_event_id = $result->getId();
            $lezione->save();
            return $result;
        }
    }
    
    public function deleteLezione($googleEventId) {
        return $this->service->events->delete('primary', $googleEventId);
    }
}
```

### Implementazione Rapida

**2. Service Class (`includes/NotificationService.php`)**
```php
<?php
namespace MusicAll\Services;

use Twilio\Rest\Client;

class NotificationService {
    
    public function notificaAssenza($assenza) {
        $lezione = $assenza->lezione;
        $allievo = $lezione->allievo;
        $docente = $lezione->docente;
        
        // Notifica al docente
        $this->invia($docente, [
            'tipo' => 'assenza_allievo',
            'messaggio' => "🚫 ASSENZA: {$allievo->nome} {$allievo->cognome} non verrà alla lezione di {$lezione->giorno_settimana} {$lezione->ora_inizio}",
            'data' => $assenza->data_assenza,
        ]);
        
        // Conferma all'allievo (opzionale)
        if ($assenza->da_recuperare) {
            $this->invia($allievo, [
                'tipo' => 'conferma_assenza',
                'messaggio' => "✅ Assenza registrata per {$lezione->giorno_settimana} {$lezione->ora_inizio}. Da recuperare.",
            ]);
        }
    }
    
    public function notificaRecupero($lezioneCustom) {
        $allievo = $lezioneCustom->allievo;
        $docente = $lezioneCustom->docente;
        
        $msg = "📅 RECUPERO PROGRAMMATO\n"
             . "Data: {$lezioneCustom->data->format('d/m/Y')}\n"
             . "Ora: {$lezioneCustom->ora_inizio}\n"
             . "Aula: {$lezioneCustom->aula->nome}\n"
             . "Materia: {$lezioneCustom->materia->nome}";
        
        $this->invia($allievo, ['tipo' => 'recupero', 'messaggio' => $msg]);
        $this->invia($docente, ['tipo' => 'recupero', 'messaggio' => $msg]);
    }
    
    public function notificaSpostamento($lezioneCustom, $motivazione) {
        $msg = "⚠️ LEZIONE SPOSTATA\n"
             . "Nuova data: {$lezioneCustom->data->format('d/m/Y')}\n"
             . "Ora: {$lezioneCustom->ora_inizio}\n"
             . "Motivo: {$motivazione}";
        
        $this->invia($lezioneCustom->allievo, ['messaggio' => $msg]);
        $this->invia($lezioneCustom->docente, ['messaggio' => $msg]);
    }
    
    private function invia($destinatario, $dati) {
        // Priorità: WhatsApp > Email > SMS
        if ($destinatario->telefono && config('notifications.whatsapp.enabled')) {
            $this->inviaWhatsApp($destinatario->telefono, $dati['messaggio']);
        } elseif ($destinatario->email) {
            $this->inviaEmail($destinatario->email, $dati);
        }
        
        // Log notifica
        DB::table('notifiche_log')->insert([
            'destinatario_tipo' => get_class($destinatario),
            'destinatario_id' => $destinatario->id,
            'canale' => 'whatsapp',
            'messaggio' => $dati['messaggio'],
            'inviato_at' => now(),
        ]);
    }
    
    private function inviaWhatsApp($telefono, $messaggio) {
        $client = new Client(
            config('notifications.whatsapp.twilio_sid'),
            config('notifications.whatsapp.twilio_token')
        );
        
        // Formatta numero italiano: +39 seguito da numero
        $to = 'whatsapp:+39' . preg_replace('/[^0-9]/', '', $telefono);
        
        $client->messages->create($to, [
            'from' => config('notifications.whatsapp.from'),
            'body' => $messaggio,
        ]);
    }
    
    private function inviaEmail($email, $dati) {
        // Usa PHPMailer o mail() nativo
        mail($email, 
             'Notifica MusicAll', 
             $dati['messaggio'],
             'From: ' . config('notifications.email.from')
        );
    }
}
```

require_once '../includes/bootstrap.php';

    ->where('message_sid', $messageSid)

        'stato' => $status,

if ($status === 'failed') {
    $notifica = DB::table('notifiche_log')

        ->first();
    
    // Fallback su email
    mail($notifica->email_destinatario,

}

http_response_code(200);
